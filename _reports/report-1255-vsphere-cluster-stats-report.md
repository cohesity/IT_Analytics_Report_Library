MAX(p.memory_size_kb/div_by) max_vm_memory,\n  AVG(p.memory_size_kb/div_by) avg_vm_memory \nFROM \n  apt_v_virtual_system  vs, apt_v_partition p, var \nWHERE \n  vs.host_id IN (${hosts})\n  AND vs.virtual_system_id = p.virtual_system_id \n  AND vs.collection_status != 3 \n  AND p.partition_type = 'VM' \n  AND p.power_state = 'poweredOn'\nGROUP BY \nvs.vmw_cluster_id\n),\nds0 AS (-- get the datastore capacity\nSELECT DISTINCT \n  vs.vmw_cluster_id, \n  ds.vmw_datastore_id, \n  ds.tot_capacity_kb/div_by total_capacity, \n  ds.free_capacity_kb/div_by free_capacity\nFROM \n  apt_v_vmw_map_vsys_dtstore md, apt_v_vmw_datastore ds, apt_v_virtual_system vs, var\nWHERE \n  vs.host_id IN (${hosts})\n  AND vs.virtual_system_id = md.virtual_system_id\n  AND md.vmw_datastore_id  = vmw_cluster_id\n),\nrr1 AS (\nSELECT DISTINCT\n  vmw_cluster_id,\n  ROUND((RATIO_TO_REPORT(ds_total_capacity) OVER ()),2) rr_ds_total_capacity\nFROM \n  ds1\n),\nvmdk AS ( --get the vmdk info\nSELECT\n  vs.vmw_cluster_id, \n  SUM(vf.file_size/div_by) total_vmdk_size,\n  MIN(vf.file_size/div_by) min_vmdk_size,\n  MAX(vf.file_size/div_by) max_vmdk_size,\n  AVG(vf.file_size/div_by)
