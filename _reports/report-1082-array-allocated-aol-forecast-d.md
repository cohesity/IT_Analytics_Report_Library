sql_query: "--Author: rich.rose@aptare.com\n--Last Modified: 03/30/2012\n--Computes the future capacity of an array based on \n--Average over Last (AOL) Last period - average period size\n--This is a drill down report that can't run stand alone\nWITH \nd0 AS (--Convert the human readable date selection to numeric values\nSELECT\nDECODE('${TheGroupBy}','Day',1,'Week',7,'Month',31,'Quarter',93,'Year',365.24) nbr_of_days,\n--Limit the forecasting periods to 48\nCASE \nWHEN to_number('${TheForecastPeriods}') > 48 THEN 48 \nELSE to_number('${TheForecastPeriods}') END forecast_periods,\n'${TheGroupBy}' the_group_by --Need this for the drilldown\nFROM apt_v_dual\n),\na0 as (--Get all the capacity merics for each period\nSELECT \nstorage_array_id,\ntrunc(log_date,DECODE('${TheGroupBy}','Day','DD','Week','WW','Month','MM','Quarter','Q','Year','YY')) log_date,\nrow_number() OVER (partition BY storage_array_id ORDER BY storage_array_id) row_number,\nmax(allocated_gb) allocated_gb,\nmax(capacity_gb) capacity_gb,\nmax(capacity_gb) - max(allocated_gb)  available_gb\nFROM aps_v_storage_array_log\nWHERE log_date BETWEEN ${startDate} AND ${endDate}\nAND array_name = '${TheArrayName}'\nGROUP BY storage_array_id,trunc(log_date,DECODE('${TheGroupBy}','Day','DD','Week','WW','Month','MM','Quarter','Q','Year','YY'))\n),\np0 AS (--Get the number of periods found, will need this to get the last period\nSELECT \nstorage_array_id,\nmin(log_date) first_period_date,\nmax(log_date) last_period_date,\nCOUNT(log_date) period_count\nFROM a0\nGROUP BY storage_array_id\n),\np1 AS (--Get the Period over Period Numbers\nSELECT\na0.storage_array_id, \nsum(DECODE(a0.row_number,1,a0.capacity_gb,0)) first_period_capacity_gb,\nsum(DECODE(a0.row_number,1,a0.allocated_gb,0)) first_period_allocated_gb,\nsum(DECODE(a0.row_number,1,a0.available_gb,0)) first_period_available_gb,\nsum(DECODE(a0.row_number,p0.period_count,a0.capacity_gb,0)) last_period_capacity_gb,\nsum(DECODE(a0.row_number,p0.period_count,a0.allocated_gb,0)) last_period_allocated_gb,\nsum(DECODE(a0.row_number,p0.period_count,a0.available_gb,0)) last_period_available_gb\nFROM a0,p0\nWHERE a0.storage_array_id = p0.storage_array_id\nAND a0.row_number IN (1,p0.period_count) --Only get the first and last period numbers\nGROUP BY a0.storage_array_id\n),\np2 AS (--Calculate the Period Over Period Growth\nSELECT\np1.storage_array_id, \np1.first_period_capacity_gb,\np1.first_period_allocated_gb,\np1.first_period_available_gb,\np1.last_period_capacity_gb,\np1.last_period_allocated_gb,\np1.last_period_available_gb,\n(p1.last_period_capacity_gb - p1.first_period_capacity_gb) / p0.period_count pop_capacity_growth,\n(p1.last_period_allocated_gb - p1.first_period_allocated_gb) / p0.period_count pop_allocated_growth,\n(p1.last_period_available_gb - p1.first_period_available_gb) / p0.period_count pop_available_growth\nFROM p1,p0\n) \nSELECT --Plot the prior history\na0.log_date log_date, \na0.capacity_gb,\na0.allocated_gb,\na0.available_gb,\n0 future_allocated_gb,\n0 future_available_gb,\np2.pop_capacity_growth,\np2.pop_allocated_growth,\np2.pop_available_growth,\np2.last_period_allocated_gb - (p2.pop_allocated_growth*(p0.period_count-rownum)) allocated_trend_line\nFROM a0,p2,p0\nWHERE a0.storage_array_id = p2.storage_array_id\nUNION ALL --Future values\nSELECT \ntrunc(the_date,DECODE('${TheGroupBy}','Day','DD','Week','WW','Month','MM','Quarter','Q','Year','YY')) log_date,\np2.last_period_capacity_gb + (p2.pop_capacity_growth*rownum) capacity_gb,\n0 allocated_gb,\n0 available_gb,\np2.last_period_allocated_gb + (p2.pop_allocated_growth*rownum) future_allocated_gb,\np2.last_period_available_gb + (p2.pop_available_growth*rownum) future_available_gb,\np2.pop_capacity_growth,\np2.pop_allocated_growth,\np2.pop_available_growth,\np2.last_period_allocated_gb + (p2.pop_allocated_growth*rownum) allocated_trend_line\nFROM a0,p0,p2,d0, TABLE(rtd.APTlistOfDates(p0.last_period_date+d0.nbr_of_days,p0.last_period_date+(d0.forecast_periods*d0.nbr_of_days),DECODE('${TheGroupBy}','Day',11,'Week',12,'Month',13,'Quarter',14,'Year',15))) the_dates\nWHERE a0.storage_array_id = p2.storage_array_id\nAND a0.storage_array_id = p0.storage_array_id\nAND a0.row_number = p0.period_count --Only need the last row"
