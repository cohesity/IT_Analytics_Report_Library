),\n  a1 AS (--Get the host disk IDs of LUNS that are mapped\n  SELECT DISTINCT \n    lm.host_disk_id,\n    sp.logical_unit_id,\n    sp.logical_unit_name,\n    sp.total_capacity_kb/1024/1024 AS sp.array_port_id\n  AND lm.scsi_os_lun_nbr  = sp.lun_nbr\n  ) ,\n  a2 AS (--Get all the physical disks that \n  SELECT \n    vc.container_name AS virtual_center_name,\n    c.vmw_cluster_id,\n    c.cluster_name,\n    pd.virtual_system_id,\n    vs.virtual_system_name,\n    h.host_name,\n    a0.vmw_datastore_id,\n    a0.datastore_name,\n    pd.canonical_name,\n    --ft.lun_nbr AS a0,\n    apt_v_virtual_system vs,\n    apt_v_vmw_cluster c,\n    --apt_vmw_fc_target ft,\n    apt_v_vmw_container vc,\n    a1,\n    aps_v_host h\n  WHERE pd.virtual_system_id  = vs.virtual_system_id\n  AND vs.virtual_center_id    = vc.virtual_center_id\n  AND vs.host_id              = h.host_id\n  AND vc.container_type       = 'virtualCenter'\n  AND vs.vmw_cluster_id       = c.vmw_cluster_id (+)\n  AND pd.vmw_physical_disk_id = a0.vmw_physical_disk_id (+)\n  --AND pd.vmw_physical_disk_id = ft.vmw_physical_disk_id (+)\n  AND pd.host_disk_id         = a1.host_disk_id (+)\n  AND aptstringconcat(DISTINCT a2.vmw_cluster_id)                                                     AS vmw_cluster_id,\n    aptstringconcat(DISTINCT a2.cluster_name)                                                       AS cluster_name,\n    upper(aptstringconcat(DISTINCT a2.virtual_system_name) )                                        AS virtual_system_group,\n    upper(aptstringconcat(DISTINCT
