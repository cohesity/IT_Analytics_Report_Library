algorithm y=mX+b\n--Period Over Period (POP) ((last period - first period) / nbr of periods) + last period...\n--First period goes back as far back as data collection began and last period is \n--the current array capacity of the last time data collection occurred. \nWITH \nd0 AS (\nSELECT\nDECODE('${freeCombo1}','Day',1,'Week',7,'Month',31,'Quarter',93,'Year',365.24) nbr_of_days,\n'${freeCombo1}' the_group_by\nFROM apt_v_dual\n),\na0 as (\nSELECT \nstorage_array_id,\n\npool_id,\ntrunc(log_date,DECODE('${freeCombo1}','Day','DD','Week','WW','Month','MM','Quarter','Q','Year','YY')) log_date,\nrow_number() over (partition by storage_array_id,  pool_id order by storage_array_id, pool_id) row_number,\nmax(capacity_kb/1024/1024) capacity,\nmax(allocated_kb/1024/1024) allocated,\nmax(unallocated_kb/1024/1024) unallocated,\nmax(available_kb/1024/1024) available,\nmax(capacity_of_vvols_kb/1024/1024) capacity_of_vvols,\nmax(touched_kb/1024/1024) touched,\nmax(associated_warn_kb/1024/1024) associated_warn,\nmax(allocated_warn_kb/1024/1024) allocated_warn\nFROM aps_v_hds_journal_pool_log \nWHERE log_date BETWEEN ${startDate} AND ${endDate}\nAND array_name = DECODE('${queryCombo1}', ' All',array_name, '${queryCombo1}')\nAND pool_function = 5\nGROUP BY storage_array_id,pool_id,trunc(log_date,DECODE('${freeCombo1}','Day','DD','Week','WW','Month','MM','Quarter','Q','Year','YY'))\n),\np0 AS (--Get the first and last period number for each array & pool \nSELECT\nstorage_array_id,\n\npool_id,\nMIN(row_number) first_period_number, --First period will always be 1\nMAX(row_number) last_period_number, --Last period is the current period or last time data was collected\nmin(log_date) first_period_date,\nmax(log_date) last_period_date,\nCOUNT(log_date) period_count\nFROM a0\nGROUP BY \nstorage_array_id,\n\npool_id\n),\npop1 AS (--Get the first and last period metrics based on the row number of the periods\nSELECT\na0.storage_array_id, a0.pool_id,\nMAX(last_period_number) nbr_of_periods,\nSUM(DECODE(a0.row_number,first_period_number,a0.available,0)) first_available,\nSUM(DECODE(a0.row_number,last_period_number,a0.available,0)) last_available\nFROM a0,p0\nWHERE a0.storage_array_id = p0.storage_array_id\nAND a0.pool_id = p0.pool_id\nGROUP BY a0.storage_array_id, a0.pool_id\n),\npop2 AS (--Calculate the Period Over Period Changes\nSELECT\nstorage_array_id, \n\npool_id,\nfirst_available,\nlast_available,\nROUND(last_available - first_available,2) pop_available_delta,\nROUND((last_available - first_available ) / DECODE(nbr_of_periods,0,null,nbr_of_periods),2) pop_available_trend\nFROM pop1\n),\nlin1 AS (--Get the Linear Growth statistics\nSELECT \na0.storage_array_id,\na0.pool_id,\nROUND(REGR_SLOPE(available,row_number),2) linear_trend,\nREGR_R2(available,row_number) r2_value,\nREGR_AVGX(available,row_number) avex,\nREGR_AVGY(available,row_number)  avey,\nREGR_INTERCEPT(available,row_number) y_intercept,\nREGR_COUNT(available,row_number) period_count,\nSTDDEV(available)  stddev\nFROM a0\nGROUP BY storage_array_id,  pool_id\n)\nSELECT \npop1.storage_array_id,\nsa.array_name,\npop1.pool_id,\npop2.first_available,\npop2.last_available,\nlin1.avey average_available,\nperiod_count,\n--\npop_available_trend,\nlinear_trend,\n--\nCASE \nWHEN period_count < 3 THEN null\nWHEN pop_available_trend > 0  THEN null\nWHEN pop_available_trend < 0  THEN round(pop1.last_available/DECODE(abs(pop_available_trend),0,null,abs(pop_available_trend)),0) \nELSE NULL\nEND pop_periods_left,\nCASE \nWHEN period_count < 3 THEN null\nWHEN linear_trend > 0  THEN null\nWHEN linear_trend < 0  THEN round(pop1.last_available/DECODE(abs(linear_trend),0,null,abs(linear_trend)),0) \nELSE NULL\nEND linear_periods_left,\nr2_value*100 r2_value,\nlin1.y_intercept,\nCASE\nWHEN pop_available_trend < 0 THEN 'red'\nWHEN pop_available_trend = 0 THEN 'white'\nWHEN pop_available_trend > 0 THEN 'green'\nEND pop_status,\nCASE\nWHEN pop_available_trend < 0 THEN 'Available Capacity Decreasing by '||abs(pop_available_trend)||'(GB) '||the_group_by||' Over '||the_group_by\nWHEN pop_available_trend = 0 THEN 'Available Capacity Staying Flat '||the_group_by||' Over '||the_group_by\nWHEN pop_available_trend > 0 THEN 'Available Capacity Increasing by '||pop_available_trend||'(GB) '||the_group_by||' Over '||the_group_by\nEND pop_status_name,\nCASE \nWHEN pop1.last_available <= 0 THEN 'Already out of capacity'\nWHEN pop_available_trend = 0 THEN 'Not changing'\nWHEN period_count < 3 THEN 'Not enough data to calculate'\nWHEN pop_available_trend > 0 THEN 'Available capacity increased'\nWHEN pop_available_trend < 0 THEN 'Will run out on:'|| to_char(sysdate+(d0.nbr_of_days*pop1.last_available/DECODE(abs(pop_available_trend),0,null,abs(pop_available_trend))),'MM/DD/YYYY')\nEND pop_date_of_concern,\nCASE\nWHEN linear_trend < 0 THEN 'red'\nWHEN linear_trend = 0 THEN 'white'\nWHEN linear_trend > 0 THEN 'green'\nEND linear_status,\nCASE\nWHEN linear_trend < 0 THEN 'Available Capacity Decreasing by '||abs(linear_trend)||'(GB) '||the_group_by||' Over '||the_group_by\nWHEN linear_trend = 0 THEN 'Available Capacity Staying Flat '||the_group_by||' Over '||the_group_by\nWHEN linear_trend > 0 THEN 'Available Capacity Increasing by '||linear_trend||'(GB) '||the_group_by||' Over '||the_group_by\nEND linear_status_name,\nCASE \nWHEN pop1.last_available <= 0 THEN 'Already out of capacity'\nWHEN linear_trend = 0 THEN 'Not changing'\nWHEN period_count < 3 THEN 'Not enough data to calculate'\nWHEN linear_trend > 0 THEN 'Available capacity increased'\nWHEN linear_trend < 0 THEN 'Will run out on:'|| to_char(sysdate+(d0.nbr_of_days*pop1.last_available/DECODE(abs(linear_trend),0,null,abs(linear_trend))),'MM/DD/YYYY')\nEND linear_date_of_concern,\nd0.nbr_of_days, \nd0.the_group_by\nFROM  d0, lin1, pop1, pop2, aps_v_storage_array sa\nWHERE\npop1.storage_array_id = sa.storage_array_id\nAND pop1.storage_array_id = pop2.storage_array_id\nAND pop1.pool_id = pop2.pool_id\nAND pop1.storage_array_id = lin1.storage_array_id\nAND pop1.pool_id = lin1.pool_id"
